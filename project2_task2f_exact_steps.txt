5) Build a convergence summary CSV from run summaries
------------------------------------------------------------
cat > /home/ds503/task2_convergence.csv << 'EOF'
label,mode,k,rounds_executed,converged,epsilon,reducers,used_combiner
EOF

for f in /home/ds503/task2_logs/task2_logs/exp_*_summary.txt; do
  label=$(basename "$f" _summary.txt)
  mode=$(grep '^mode=' "$f" | cut -d= -f2)
  k=$(grep '^k=' "$f" | cut -d= -f2)
  rounds=$(grep '^rounds_executed=' "$f" | cut -d= -f2)
  conv=$(grep '^converged=' "$f" | cut -d= -f2)
  eps=$(grep '^epsilon=' "$f" | cut -d= -f2)
  red=$(grep '^reducers=' "$f" | cut -d= -f2)
  comb=$(grep '^used_combiner=' "$f" | cut -d= -f2)
  echo "$label,$mode,$k,$rounds,$conv,$eps,$red,$comb" >> /home/ds503/task2_convergence.csv
done
