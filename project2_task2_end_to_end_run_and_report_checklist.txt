5) Task 2.2(f) - Experiments (K and R variations) (inside container)
============================================================

# 5.1 Create experiment timing file
echo "timestamp,label,time_ms,time_s,success" > /home/ds503/task2_exp_times.csv

# 5.2 Run matrix for K in {3,5,8} and R in {10,30} using EARLY and OPT_CENTERS
for K in 3 5 8; do
  for R in 10 30; do
    for MODE in EARLY OPT_CENTERS; do
      OUT=/project2/task2/exp/k${K}_r${R}_${MODE,,}
      hdfs dfs -rm -r -f "$OUT" >/dev/null 2>&1 || true

      start=$(date +%s%3N)
      if hadoop jar /home/ds503/ds503_bdm-1.0-SNAPSHOT.jar project2.task2simple.KMeansMain \
          /home/ds503/project2_task2_config.properties \
          mode=$MODE k=$K rounds=$R output.base=$OUT > /home/ds503/task2_logs/exp_k${K}_r${R}_${MODE,,}.log 2>&1; then
          rc=0
      else
          rc=1
      fi
      end=$(date +%s%3N)
      ms=$((end-start))
      sec=$(awk -v ms="$ms" 'BEGIN{printf "%.3f", ms/1000.0}')
      ts=$(date -u +"%Y-%m-%dT%H:%M:%S.000")
      echo "$ts,exp_k${K}_r${R}_${MODE,,},$ms,$sec,$([ $rc -eq 0 ] && echo 1 || echo 0)" >> /home/ds503/task2_exp_times.csv

      hdfs dfs -cat "$OUT/summary.txt" > /home/ds503/task2_logs/exp_k${K}_r${R}_${MODE,,}_summary.txt
    done
  done
done

# 5.3 (Optional but recommended) Add one strict convergence test with higher rounds + practical epsilon
OUT=/project2/task2/exp/early_k5_r50_eps50
