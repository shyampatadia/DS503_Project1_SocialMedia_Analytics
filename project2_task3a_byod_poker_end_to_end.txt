  https://archive.ics.uci.edu/dataset/158/poker+hand

8) Run Task 2.2(a-e) equivalents on BYOD

Inside container:
mkdir -p /home/ds503/byod_poker/artifacts/logs
echo "label,time_s,success" > /home/ds503/byod_poker/artifacts/byod_times.csv

run_km () {
  label="$1"; shift
  start=$(date +%s)
  hadoop jar /home/ds503/ds503_bdm-1.0-SNAPSHOT.jar project2.task2simple.KMeansMain \
    /home/ds503/project2_byod_poker_config.properties "$@" \
    > "/home/ds503/byod_poker/artifacts/logs/${label}.log" 2>&1
  rc=$?
  end=$(date +%s)
  sec=$((end-start))
  if [ $rc -eq 0 ]; then ok=1; else ok=0; fi
  echo "$label,$sec,$ok" >> /home/ds503/byod_poker/artifacts/byod_times.csv
  echo "$label done (success=$ok, sec=$sec)"
  return $rc
}
