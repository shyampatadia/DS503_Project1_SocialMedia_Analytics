# helper (simple + reusable)
run_km () {
  label="$1"; shift
  start=$(date +%s)
  hadoop jar /home/ds503/ds503_bdm-1.0-SNAPSHOT.jar project2.task2simple.KMeansMain \
    /home/ds503/project2_task2_config.properties "$@" \
    > "/home/ds503/task2_report_plus/logs/${label}.log" 2>&1
  rc=$?
  end=$(date +%s)
  sec=$((end-start))
  echo "$label,$sec,$([ $rc -eq 0 ] && echo 1 || echo 0)" >> /home/ds503/task2_report_plus/tables/runtime.csv
  echo "$label done (sec=$sec, rc=$rc)"
}
